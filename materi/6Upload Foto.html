1 Pendahuluan

Pada jobsheet ini, kita akan menambahkan fitur Upload Foto pada CRUD Mahasiswa yang telah dibuat sebelumnya. Fitur ini memungkinkan pengguna untuk mengunggah foto profil mahasiswa saat menambah atau mengedit data.

Foto yang diupload akan disimpan di folder uploads/foto/ dengan nama file yang unik (NIM + timestamp), dan nama file akan disimpan di database pada kolom foto.

🎯 Tujuan Pembelajaran:
  • Memahami konsep upload file di CodeIgniter 3
  • Mampu mengkonfigurasi library upload
  • Mampu membuat form dengan enctype multipart/form-data
  • Mampu menampilkan preview foto sebelum upload
  • Mampu menghapus file foto saat data dihapus
  • Mampu mengupdate foto dengan menghapus foto lama

  • CodeIgniter 3 sudah terinstall
  • Sudah menyelesaikan Jobsheet 2 (CRUD Mahasiswa)

5 Model M_mhs

File: application/models/M_mhs.php

Model ini menangani semua operasi database untuk tabel mhs.

PHP 
<?php

class M_mhs extends CI_Model {
    
    public function __construct() {
        parent::__construct();
        $this->load->database();
    }
    
    // READ - Get all mahasiswa
    public function get_all_mhs() {
        return $this->db->get('mhs')->result();
    }
    
    // READ - Search mahasiswa
    public function search_mhs($keyword) {
        $this->db->group_start();
        $this->db->like('nim', $keyword);
        $this->db->or_like('nama', $keyword);
        $this->db->or_like('prodi', $keyword);
        $this->db->or_like('jurusan', $keyword);
        $this->db->group_end();
        return $this->db->get('mhs')->result();
    }
    
    // READ - Get mahasiswa by NIM
    public function get_mhs_by_nim($nim) {
        return $this->db->get_where('mhs', ['nim' => $nim])->row();
    }
    
    // CREATE - Insert mahasiswa
    public function insert_mhs($data) {
        return $this->db->insert('mhs', $data);
    }
    
    // UPDATE - Update mahasiswa
    public function update_mhs($nim, $data) {
        $this->db->where('nim', $nim);
        return $this->db->update('mhs', $data);
    }
    
    // DELETE - Delete mahasiswa
    public function delete_mhs($nim) {
        $this->db->where('nim', $nim);
        return $this->db->delete('mhs');
    }
    
    // HELPER - Check if NIM exists
    public function nim_exists($nim) {
        $this->db->where('nim', $nim);
        $query = $this->db->get('mhs');
        return $query->num_rows() > 0;
    }
}
📌 Penjelasan Model:
  • Tidak ada perubahan signifikan dari Jobsheet 2

  • Semua method tetap sama karena kolom foto hanya tambahan

6 Controller Mhs

File: application/controllers/Mhs.php

Controller ini menangani semua operasi CRUD Mahasiswa dengan upload foto.

PHP 
<?php

class Mhs extends CI_Controller {
    
    public function __construct() {
        parent::__construct();
        $this->load->model('M_mhs');
        $this->load->helper('form');
        $this->load->helper('file');
        $this->load->library('form_validation');
        $this->load->library('upload');
        
        if(!$this->session->userdata('logged_in')) {
            redirect('auth');
        }
    }
    
    // ===== READ =====
    public function index() {
        $keyword = $this->input->get('search');
        
        if (!empty($keyword)) {
            $data['mhs'] = $this->M_mhs->search_mhs($keyword);
            $data['keyword'] = $keyword;
        } else {
            $data['mhs'] = $this->M_mhs->get_all_mhs();
            $data['keyword'] = '';
        }
        
        $this->load->view('dashboard/mhs/index', $data);
    }
    
    // ===== CREATE =====
    public function add() {
        $this->load->view('dashboard/mhs/add');
    }
    
    public function save() {
        // Validasi form
        $this->form_validation->set_rules('nim', 'NIM', 'required|min_length[5]|max_length[20]');
        $this->form_validation->set_rules('nama', 'Nama', 'required|min_length[3]|max_length[100]');
        $this->form_validation->set_rules('prodi', 'Prodi', 'required|min_length[3]|max_length[50]');
        $this->form_validation->set_rules('jurusan', 'Jurusan', 'required|min_length[3]|max_length[50]');
        
        $this->form_validation->set_message('required', '{field} harus diisi!');
        $this->form_validation->set_message('min_length', '{field} minimal {param} karakter!');
        $this->form_validation->set_message('max_length', '{field} maksimal {param} karakter!');
        
        if ($this->form_validation->run() == FALSE) {
            $this->session->set_flashdata('error', validation_errors());
            redirect('mhs/add');
        } else {
            $nim = $this->input->post('nim');
            
            // Cek NIM duplikat
            if ($this->M_mhs->nim_exists($nim)) {
                $this->session->set_flashdata('error', 'NIM ' . $nim . ' sudah terdaftar!');
                redirect('mhs/add');
            }
            
            // Upload foto
            $foto_name = $this->upload_foto($nim);
            
            $data = [
                'nim' => $nim,
                'nama' => $this->input->post('nama'),
                'prodi' => $this->input->post('prodi'),
                'jurusan' => $this->input->post('jurusan'),
                'foto' => $foto_name
            ];
            
            if ($this->M_mhs->insert_mhs($data)) {
                $this->session->set_flashdata('success', 'Data mahasiswa berhasil ditambahkan!');
                redirect('mhs');
            }
        }
    }
    
    // ===== UPDATE =====
    public function edit($nim) {
        $data['mhs'] = $this->M_mhs->get_mhs_by_nim($nim);
        
        if (empty($data['mhs'])) {
            $this->session->set_flashdata('error', 'Data mahasiswa tidak ditemukan!');
            redirect('mhs');
        }
        
        $this->load->view('dashboard/mhs/edit', $data);
    }
    
    public function update($nim) {
        $mhs = $this->M_mhs->get_mhs_by_nim($nim);
        if (empty($mhs)) {
            $this->session->set_flashdata('error', 'Data mahasiswa tidak ditemukan!');
            redirect('mhs');
        }
        
        // Validasi form
        $this->form_validation->set_rules('nim', 'NIM', 'required|min_length[5]|max_length[20]');
        $this->form_validation->set_rules('nama', 'Nama', 'required|min_length[3]|max_length[100]');
        $this->form_validation->set_rules('prodi', 'Prodi', 'required|min_length[3]|max_length[50]');
        $this->form_validation->set_rules('jurusan', 'Jurusan', 'required|min_length[3]|max_length[50]');
        
        $this->form_validation->set_message('required', '{field} harus diisi!');
        $this->form_validation->set_message('min_length', '{field} minimal {param} karakter!');
        $this->form_validation->set_message('max_length', '{field} maksimal {param} karakter!');
        
        if ($this->form_validation->run() == FALSE) {
            $this->session->set_flashdata('error', validation_errors());
            redirect('mhs/edit/' . $nim);
        } else {
            $new_nim = $this->input->post('nim');
            
            // Cek NIM duplikat jika berubah
            if ($new_nim != $nim && $this->M_mhs->nim_exists($new_nim)) {
                $this->session->set_flashdata('error', 'NIM ' . $new_nim . ' sudah terdaftar!');
                redirect('mhs/edit/' . $nim);
            }
            
            // Upload foto baru
            $foto_name = $this->upload_foto($new_nim, $mhs->foto);
            
            $data = [
                'nim' => $new_nim,
                'nama' => $this->input->post('nama'),
                'prodi' => $this->input->post('prodi'),
                'jurusan' => $this->input->post('jurusan'),
                'foto' => $foto_name
            ];
            
            if ($this->M_mhs->update_mhs($nim, $data)) {
                $this->session->set_flashdata('success', 'Data mahasiswa berhasil diupdate!');
                redirect('mhs');
            }
        }
    }
    
    // ===== DELETE =====
    public function confirm_delete($nim) {
        $data['mhs'] = $this->M_mhs->get_mhs_by_nim($nim);
        
        if (empty($data['mhs'])) {
            $this->session->set_flashdata('error', 'Data mahasiswa tidak ditemukan!');
            redirect('mhs');
        }
        
        $this->load->view('dashboard/mhs/confirm_delete', $data);
    }
    
    public function delete($nim) {
        $mhs = $this->M_mhs->get_mhs_by_nim($nim);
        
        if (empty($mhs)) {
            $this->session->set_flashdata('error', 'Data mahasiswa tidak ditemukan!');
            redirect('mhs');
        }
        
        // Hapus foto jika ada
        if (!empty($mhs->foto)) {
            $foto_path = FCPATH . 'uploads/foto/' . $mhs->foto;
            if (file_exists($foto_path)) {
                unlink($foto_path);
            }
        }
        
        if ($this->M_mhs->delete_mhs($nim)) {
            $this->session->set_flashdata('success', 'Data mahasiswa berhasil dihapus!');
        }
        
        redirect('mhs');
    }
    
    // ===== PRIVATE METHOD UPLOAD =====
    private function upload_foto($nim, $old_foto = null) {
        // Cek apakah ada file yang diupload
        if (empty($_FILES['foto']['name'])) {
            return $old_foto;
        }
        
        // Path folder upload
        $upload_path = FCPATH . 'uploads/foto/';
        
        // Buat folder jika belum ada
        if (!is_dir($upload_path)) {
            mkdir($upload_path, 0777, true);
        }
        
        // Konfigurasi upload
        $config['upload_path'] = $upload_path;
        $config['allowed_types'] = 'jpg|jpeg|png|gif';
        $config['max_size'] = 2048;
        $config['file_name'] = $nim . '_' . time();
        $config['overwrite'] = true;
        
        $this->upload->initialize($config);
        
        if ($this->upload->do_upload('foto')) {
            $upload_data = $this->upload->data();
            $foto_name = $upload_data['file_name'];
            
            // Hapus foto lama jika ada
            if (!empty($old_foto)) {
                $old_path = $upload_path . $old_foto;
                if (file_exists($old_path) && $old_foto != $foto_name) {
                    unlink($old_path);
                }
            }
            
            return $foto_name;
        } else {
            $error = $this->upload->display_errors('', '');
            $this->session->set_flashdata('error', 'Upload foto gagal: ' . $error);
            return $old_foto;
        }
    }
}

  • ✅ Hapus foto saat data dihapus

11 Alur Upload Foto

Tambah/Edit
→
Pilih File
→
Preview
→
Submit
→
Upload
→
Simpan
→
Selesai
📌 Penjelasan Alur Upload:
  • 1. Pilih File - User memilih file gambar melalui input file
  • 2. Preview - Gambar ditampilkan secara real-time menggunakan JavaScript
  • 3. Submit - User mengklik tombol Simpan/Update
  • 4. Upload - File diupload ke server menggunakan library upload CI
  • 5. Simpan - Nama file disimpan ke database, file disimpan di folder uploads/foto/
  • 6. Selesai - Redirect ke halaman daftar mahasiswa
✅ Validasi Upload:
  • File harus berupa gambar (jpg, jpeg, png, gif)
  • Ukuran file maksimal 2MB
  • Nama file unik (NIM + timestamp)
  • Folder upload dibuat otomatis jika belum ada
⚠️ Penanganan Error:
  • Jika upload gagal → tampilkan flashdata error
  • Jika folder tidak bisa ditulis → tampilkan error
  • Jika file terlalu besar → tampilkan error
  • Jika format tidak diizinkan → tampilkan error
