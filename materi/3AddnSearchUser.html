3 Database

Buat database dengan nama ci_app dan tabel users:

SQL 
CREATE DATABASE ci_app;
USE ci_app;

CREATE TABLE users (
    id INT(11) PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(32) NOT NULL,
    role ENUM('admin', 'dosen', 'mahasiswa') DEFAULT 'mahasiswa',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Contoh data untuk testing
INSERT INTO users (username, password, role) VALUES 
('admin', MD5('admin123'), 'admin'),
('dosen', MD5('dosen123'), 'dosen'),
('mahasiswa', MD5('mahasiswa123'), 'mahasiswa');
✅ Catatan:
  • Password menggunakan MD5 (32 karakter)
  • Role: admin, dosen, atau mahasiswa
  • Username harus UNIQUE

4 Controller Auth

File: application/controllers/Auth.php

Controller ini menangani proses login dan logout.

PHP 
<?php

class Auth extends CI_Controller {
    
    public function __construct() {
        parent::__construct();
        $this->load->model('M_user');
    }
    
    public function index() { 
        if($this->session->userdata('logged_in')) {
            redirect('dashboard');
        }
        $this->load->view('login'); 
    }

    public function login_action() {
        $username = $this->input->post('username');
        $password = md5($this->input->post('password'));
        
        $user = $this->M_user->get_user_by_username($username);

        if ($user && $user->password == $password) {
            $this->session->set_userdata([
                'user_id' => $user->id,
                'username' => $user->username,
                'role' => $user->role, 
                'logged_in' => TRUE
            ]);
            redirect('dashboard');
        } else {
            $this->session->set_flashdata('error', 'Username atau Password salah!');
            redirect('Auth');
        }
    }

    public function logout() {
        $this->session->sess_destroy();
        redirect('auth');
    }
}

5 Controller Dashboard

File: application/controllers/Dashboard.php

Controller ini menangani semua operasi CRUD Users.

PHP 
<?php

class Dashboard extends CI_Controller {
    
    public function __construct() {
        parent::__construct();
        $this->load->model('M_user');
        $this->load->helper('form');
        $this->load->library('form_validation');
        
        if(!$this->session->userdata('logged_in')) {
            redirect('auth');
        }
    }
    
    // ===== READ =====
    public function index() {
        $keyword = $this->input->get('search');
        
        if (!empty($keyword)) {
            $data['users'] = $this->M_user->search_users($keyword);
            $data['keyword'] = $keyword;
        } else {
            $data['users'] = $this->M_user->get_all_users();
            $data['keyword'] = '';
        }
        
        $this->load->view('dashboard/admin', $data);
    }
    
    // ===== CREATE =====
    public function add() {
        $data['title'] = 'Tambah User';
        $this->load->view('dashboard/add_user', $data);
    }
    
    public function save() {
        $this->form_validation->set_rules('username', 'Username', 'required|min_length[3]|max_length[50]|is_unique[users.username]');
        $this->form_validation->set_rules('password', 'Password', 'required|min_length[4]');
        $this->form_validation->set_rules('confirm_password', 'Konfirmasi Password', 'required|matches[password]');
        $this->form_validation->set_rules('role', 'Role', 'required|in_list[admin,dosen,mahasiswa]');
        
        if ($this->form_validation->run() == FALSE) {
            $this->session->set_flashdata('error', validation_errors());
            redirect('dashboard/add');
        } else {
            $data = [
                'username' => $this->input->post('username'),
                'password' => md5($this->input->post('password')),
                'role' => $this->input->post('role')
            ];
            
            if ($this->M_user->insert_user($data)) {
                $this->session->set_flashdata('success', 'User berhasil ditambahkan!');
                redirect('dashboard');
            }
        }
    }
    
    // ===== UPDATE =====
    public function edit($id) {
        $data['user'] = $this->M_user->get_user_by_id($id);
        if (empty($data['user'])) {
            $this->session->set_flashdata('error', 'User tidak ditemukan!');
            redirect('dashboard');
        }
        $this->load->view('dashboard/edit_user', $data);
    }
    
    public function update($id) {
        $user = $this->M_user->get_user_by_id($id);
        if (empty($user)) {
            $this->session->set_flashdata('error', 'User tidak ditemukan!');
            redirect('dashboard');
        }
        
        $this->form_validation->set_rules('username', 'Username', 'required|min_length[3]|max_length[50]');
        $this->form_validation->set_rules('role', 'Role', 'required|in_list[admin,dosen,mahasiswa]');
        
        if ($this->input->post('password') != '') {
            $this->form_validation->set_rules('password', 'Password', 'min_length[4]');
            $this->form_validation->set_rules('confirm_password', 'Konfirmasi Password', 'matches[password]');
        }
        
        if ($this->form_validation->run() == FALSE) {
            $this->session->set_flashdata('error', validation_errors());
            redirect('dashboard/edit/' . $id);
        } else {
            $data = [
                'username' => $this->input->post('username'),
                'role' => $this->input->post('role')
            ];
            
            if ($this->input->post('password') != '') {
                $data['password'] = md5($this->input->post('password'));
            }
            
            if ($this->M_user->update_user($id, $data)) {
                $this->session->set_flashdata('success', 'User berhasil diupdate!');
                redirect('dashboard');
            }
        }
    }
    
    // ===== DELETE =====
    public function confirm_delete($id) {
        $data['user'] = $this->M_user->get_user_by_id($id);
        if (empty($data['user'])) {
            $this->session->set_flashdata('error', 'User tidak ditemukan!');
            redirect('dashboard');
        }
        if ($id == $this->session->userdata('user_id')) {
            $this->session->set_flashdata('error', 'Tidak bisa menghapus akun sendiri!');
            redirect('dashboard');
        }
        $this->load->view('dashboard/confirm_delete', $data);
    }
    
    public function delete($id) {
        $user = $this->M_user->get_user_by_id($id);
        if (empty($user)) {
            $this->session->set_flashdata('error', 'User tidak ditemukan!');
            redirect('dashboard');
        }
        if ($id == $this->session->userdata('user_id')) {
            $this->session->set_flashdata('error', 'Tidak bisa menghapus akun sendiri!');
            redirect('dashboard');
        }
        if ($this->M_user->delete_user($id)) {
            $this->session->set_flashdata('success', 'User berhasil dihapus!');
        }
        redirect('dashboard');
    }
}

6 Model M_user

File: application/models/M_user.php

Model ini menangani semua operasi database untuk tabel users.

PHP 
<?php

class M_user extends CI_Model {
    
    public function __construct() {
        parent::__construct();
        $this->load->database();
    }
    
    // READ - Get all users
    public function get_all_users() {
        return $this->db->get('users')->result();
    }
    
    // READ - Search users
    public function search_users($keyword) {
        $this->db->like('username', $keyword);
        return $this->db->get('users')->result();
    }
    
    // READ - Get user by ID
    public function get_user_by_id($id) {
        return $this->db->get_where('users', ['id' => $id])->row();
    }
    
    // READ - Get user by username
    public function get_user_by_username($username) {
        return $this->db->get_where('users', ['username' => $username])->row();
    }
    
    // CREATE - Insert new user
    public function insert_user($data) {
        return $this->db->insert('users', $data);
    }
    
    // UPDATE - Update user
    public function update_user($id, $data) {
        $this->db->where('id', $id);
        return $this->db->update('users', $data);
    }
    
    // DELETE - Delete user
    public function delete_user($id) {
        $this->db->where('id', $id);
        return $this->db->delete('users');
    }
    
    // Count users by role
    public function count_users_by_role($role = null) {
        if ($role) {
            $this->db->where('role', $role);
        }
        return $this->db->count_all_results('users');
    }
}

            <div class="btn-group">
                <a href="<?= base_url('dashboard/delete/'.$user->id) ?>" class="btn-yes">
                    <i class="fas fa-trash"></i> Ya, Hapus!
                </a>
                <a href="<?= base_url('dashboard') ?>" class="btn-no">
                    <i class="fas fa-times"></i> Batal
                </a>
            </div>
        </div>
    </div>

</body>
</html>
📌 Alur Hapus Tanpa JavaScript:
  • User klik tombol "Hapus" di dashboard
  • Redirect ke halaman confirm_delete/{id}
  • Tampilkan halaman konfirmasi dengan data user
  • User klik "Ya, Hapus!" atau "Batal"
  • Jika "Ya, Hapus!" → redirect ke delete/{id} → hapus data
  • Tampilkan flashdata sukses/gagal → redirect ke dashboard

Berikut adalah ringkasan fitur CRUD yang telah dibuat:

Fitur URL Method Keterangan
📋 List /dashboard GET Menampilkan semua user
🔍 Search /dashboard?search=keyword GET Mencari user berdasarkan username
➕ Tambah /dashboard/add GET Form tambah user
💾 Simpan /dashboard/save POST Proses simpan user baru
✏️ Edit /dashboard/edit/{id} GET Form edit user
🔄 Update /dashboard/update/{id} POST Proses update user
🗑️ Hapus /dashboard/confirm_delete/{id} GET Konfirmasi hapus user

❌ Sebelum (Tanpa Model)

  • Query SQL di dalam controller
  • Controller menjadi "gemuk"
  • Sulit di-maintenance
  • Kode tidak reusable
  • Tidak menerapkan MVC dengan baik

✅ Sesudah (Dengan Model)

  • Query SQL di dalam model
  • Controller lebih "ramping"
  • Mudah di-maintenance
  • Kode reusable
  • Menerapkan MVC dengan baik
🎉 Keuntungan Menggunakan Model:
  • Kode lebih terstruktur dan rapi
  • Mudah untuk debugging
  • Dapat digunakan kembali di controller lain
  • Memisahkan logika bisnis dari controller
  • Memudahkan proses maintenance dan pengembangan
  • Sidebar terpisah sehingga mudah diubah
📌 Poin Penting:
  • Selalu load model di __construct() controller
  • Gunakan htmlspecialchars() untuk mencegah XSS
  • Simpan session setelah login berhasil
  • Sidebar dipanggil dengan $this->load->view('layout/sidebar')
  • CSS dan JS untuk sidebar berada di file sidebar.php
  • Role yang tersedia: admin, dosen, mahasiswa
